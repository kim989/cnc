import os
import math
import wave
import struct
import time
import threading
import tkinter as tk
from tkinter import ttk

try:
    import pygame
    HAS_PYGAME = True
except ImportError:
    HAS_PYGAME = False
    import winsound

# ==============================================================================
# 1. 비프음 WAV 파일 자동 생성 함수 (외부 음원이 없을 때 자동 창작)
# ==============================================================================
def generate_default_alarm_sound(file_path="alarm_sound.wav"):
    if os.path.exists(file_path):
        return file_path
    
    sample_rate = 44100
    duration = 0.5  # 0.5초 비프음
    freq = 880.0    # 880Hz (경고 A5 톤)
    
    with wave.open(file_path, 'w') as wav:
        wav.setnchannels(1)
        wav.setsampwidth(2)
        wav.setframerate(sample_rate)
        for i in range(int(sample_rate * duration)):
            t = float(i) / sample_rate
            val = int(32767 * 0.5 * math.sin(2.0 * math.pi * freq * t))
            wav.writeframes(struct.pack('<h', val))
    return file_path


# ==============================================================================
# 2. 알람 및 팝업 관리자 클래스
# ==============================================================================
class AlarmPopupManager:
    def __init__(self, master):
        self.master = master
        self.is_playing = False
        self.volume = 0.05      # 시작 음량 (5%)
        self.max_volume = 1.0   # 최대 음량 (100%)
        self.step_volume = 0.05 # 0.5초마다 5%씩 점진적 증가
        
        # Pygame 또는 Winsound 오디오 초기화
        if HAS_PYGAME:
            pygame.mixer.init()
            self.sound_file = generate_default_alarm_sound()
            self.sound = pygame.mixer.Sound(self.sound_file)
        else:
            self.sound = None
        
    def start_alarm(self, title="⚠️ 긴급 바이트 마모 경고", message="바이트(공구) 마모 한계 초과 감지!\n이송 속도를 낮추고 공구 팁을 교체하십시오."):
        """알람 시작 및 팝업 띄우기"""
        if self.is_playing:
            return  # 이미 알람이 울리고 있다면 중복 실행 방지
            
        self.is_playing = True
        self.volume = 0.05
        
        # 1. 백그라운드 스레드로 경고음 증폭 및 무한 루프 재생 실행 (UI 멈춤 방지)
        self.sound_thread = threading.Thread(target=self._play_sound_loop, daemon=True)
        self.sound_thread.start()
        
        # 2. 메인 UI 스레드에서 팝업창 생성
        self._create_popup_window(title, message)

    def _play_sound_loop(self):
        """소리 반복 재생 및 볼륨 점진적 증폭 백그라운드 로직"""
        if HAS_PYGAME and self.sound:
            channel = self.sound.play(loops=-1)  # -1: OFF 누르기 전까지 무한 반복
            channel.set_volume(self.volume)
            
            while self.is_playing:
                time.sleep(0.5)  # 0.5초 주기
                if not self.is_playing:
                    break
                    
                # 볼륨을 점점 커지게 설정
                if self.volume < self.max_volume:
                    self.volume = min(self.max_volume, self.volume + self.step_volume)
                    channel.set_volume(self.volume)
                    
            # OFF 시 즉시 소리 중단
            channel.stop()
        else:
            # winsound 880Hz 비프음 백그라운드 재생 (Windows 표준)
            while self.is_playing:
                try:
                    winsound.Beep(880, 400)  # 880Hz 0.4초 소리
                except Exception:
                    pass
                time.sleep(0.1)

    def _create_popup_window(self, title, message):
        """경고 팝업 UI 창 생성"""
        popup = tk.Toplevel(self.master)
        popup.title(title)
        popup.geometry("400x220")
        popup.attributes('-topmost', True)  # 화면 최상단에 강제 고정
        popup.protocol("WM_DELETE_WINDOW", lambda: self.stop_alarm(popup))  # 창 X 버튼 클릭 시에도 OFF 로직 실행
        
        # 화면 중앙 배치
        popup.update_idletasks()
        x = (popup.winfo_screenwidth() // 2) - (popup.winfo_width() // 2)
        y = (popup.winfo_screenheight() // 2) - (popup.winfo_height() // 2)
        popup.geometry(f"+{x}+{y}")

        frame = ttk.Frame(popup, padding=20)
        frame.pack(fill=tk.BOTH, expand=True)

        lbl_icon = ttk.Label(frame, text="🚨", font=("Arial", 28))
        lbl_icon.pack(pady=2)

        lbl_msg = ttk.Label(frame, text=message, font=("맑은 고딕", 11, "bold"), foreground="red", justify="center")
        lbl_msg.pack(pady=10)

        # OFF 버튼 (누르기 전까지 절대로 꺼지지 않음)
        btn_off = tk.Button(
            frame, 
            text="OFF (경고 끄기)", 
            font=("맑은 고딕", 12, "bold"),
            bg="#dc3545", 
            fg="white", 
            activebackground="#bd2130",
            activeforeground="white",
            command=lambda: self.stop_alarm(popup)
        )
        btn_off.pack(fill=tk.X, pady=10)

    def stop_alarm(self, popup_window=None):
        """알람 및 소리 끄기 인터락"""
        self.is_playing = False
        if popup_window:
            popup_window.destroy()


# ==============================================================================
# 3. 메인 실행부
# ==============================================================================
if __name__ == "__main__":
    root = tk.Tk()
    root.title("모니터링 관제 시스템")
    root.geometry("400x200")
    
    alarm_manager = AlarmPopupManager(root)

    btn_trigger = tk.Button(
        root, 
        text="🚨 이상 감지 테스트 (알람 발생)", 
        font=("맑은 고딕", 11, "bold"),
        bg="#ffc107", 
        command=alarm_manager.start_alarm
    )
    btn_trigger.pack(expand=True, ipadx=10, ipady=5)

    root.mainloop()
